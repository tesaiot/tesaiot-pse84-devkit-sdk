1/*******************************************************************************
2 * File: bento_kit.h
3 *
4 * The board's own name for itself, in one place.
5 *
6 * Two files used to derive this independently - nus_commands.c for
7 * bento.info.board and bento_fw.c for bento.fw.query - and both wrote the same
8 * two-way test: Eva Kit, or else "ai". A third board then arrived. The HMI Kit
9 * defines USE_KIT_PSE84_HMI, matched neither arm, fell through the else, and
10 * told every desktop it was an AI Kit. Confirmed on hardware 2026-09-02: both
11 * verbs answered kit:"ai" from an HMI Kit, and the desktop uses that field to
12 * decide which sensor tiles to offer - tiles for parts this board does not have.
13 *
14 * Adding a board now means adding one arm here. A board that forgets gets
15 * "unknown", which is wrong in a way somebody will notice, rather than wrong
16 * in a way that looks like a different product.
17 ******************************************************************************/
18#ifndef BENTO_KIT_H
19#define BENTO_KIT_H
20
21#if defined(USE_KIT_PSE84_EVAL_EPC2)
22#define BENTO_KIT_TAG "eva"
23#elif defined(USE_KIT_PSE84_HMI)
24#define BENTO_KIT_TAG "hmi"
25#elif defined(USE_KIT_PSE84_AI)
26#define BENTO_KIT_TAG "ai"
27#else
28/* Deliberately not a real board. A build that lands here has not declared
29 * itself, and saying so beats impersonating whichever board is listed first. */
30#define BENTO_KIT_TAG "unknown"
31#endif
32
33#endif /* BENTO_KIT_H */
