Detailed Description

9 ฟังก์ชันที่ทำงานบนไฟล์ LittleFS 2 ไฟล์: /.bentoclaw_session ซึ่งเป็น ring buffer แบบ JSONL เก็บรอบสนทนาล่าสุด CLAW_SESSION_MAX_MSGS (10) รอบ และ /.bentoclaw_memory ซึ่งเป็นที่เก็บคีย์/ค่าแบบถาวรที่อยู่รอด claw_session_clear() ตัว implementation คือ claw_session.c คอมไพล์รวมอยู่ใน libbento_mpy.a และไม่ได้ส่งมอบมาเป็นซอร์ส; มันทำ I/O กับไฟล์ด้วยการรัน open() ของ Python ผ่าน VFS bridge ของ MicroPython ด้วยเหตุนี้ทุกฟังก์ชันในกลุ่มนี้จึงต้องอยู่ใน MicroPython task context และระบบย่อยนี้จึงไม่มีการจับ lock เลย — มีเพียง MicroPython task เท่านั้นที่แตะมัน

claw_session_init() ทำงานครั้งเดียวจาก bentoclaw_ensure_init() ที่เป็น idempotent (ดู Claw safety gates); ไม่มีการเริ่มต้นระบบทางอื่นเปิดให้เรียก

6 ใน 9 ตัวมี call site อยู่จริงใน modbentoclaw.c ส่วนอีก 3 ตัว คือ claw_memory_get(), claw_session_dirty(), claw_session_flush() ไม่มีผู้เรียกที่ใดเลย ทั้งในเทมเพลตและในทรีที่เก็บไว้ใน archive; ตัวอย่างของทั้งสามเป็นตัวอย่างที่เขียนขึ้นเองและกำกับไว้เช่นนั้น

variant ที่ใช้ได้
mtb-mpy เท่านั้น (ไม่มีการลิงก์ libbento_mpy.a เมื่อ BENTO_HAS_MPY=0)

static void bentoclaw_ensure_init(void) {
if (s_module_inited) return;
claw_rate_init();
claw_cb_init();
claw_session_init();
/* Set per-tool rate limits from registry */
for (size_t i = 0; i < TOOL_COUNT; i++) {
if (s_tools[i].rate_per_min > 0) {

}
}

ข้อกำหนดการเรียกใช้
เรนเดอร์รอบสนทนาล่าสุดของเซสชันพร้อมกับ memory ที่คงอยู่ถาวร ออกมาเป็นสตริงบริบทของ prompt ผู้เรียกเป็นฝ่ายจัดหาบัฟเฟอร์ — ตัวแปร static ที่ใช้ร่วมกันของโมดูลในของที่ส่งมอบจริงคือ s_work_large[3072] และ s_work_small[512] — และการเรียกจริงส่งค่า sizeof(s_work_large) - 256 เข้าไป ซึ่งเป็นการกันพื้นที่เผื่อไว้ 256 ไบต์โดยเจตนา คืนค่าเป็นจำนวนไบต์ที่เขียนได้ ผลลัพธ์ ยังไม่ผ่านการ escape แบบ JSON: ต้องส่งผ่าน json_escape ก่อนเข้าไปอยู่ในเนื้อคำขอ ตรงตามที่ตัวอย่างอ้างอิงทำไว้ทุกประการ เป้าหมายของการ escape กับบัฟเฟอร์ที่ใช้ประกอบ JSON ไม่ทับซ้อนกันเลย (บริบทอยู่ใน s_work_large ส่วนที่ escape แล้วอยู่ใน s_work_small แล้วจึงประกอบ JSON กลับลงใน s_work_large หลังจากใช้บริบทเสร็จแล้ว) ยังมีรูปแบบที่ง่ายกว่านี้ในของที่ส่งมอบจริงซึ่งส่งขนาดบัฟเฟอร์เต็มเข้าไป ต้องเรียก claw_session_init() มาก่อน; ทำงานใน MicroPython task context; ไม่มีการจับ lock

ประกอบบริบท, escape, แล้วประกอบ JSON พร้อมการกันพื้นที่เผื่อและวินัยเรื่องบัฟเฟอร์ที่ไม่ทับซ้อนกัน
/* ...context: inside bentoclaw_ask() ... */
/* Build context → escape both → build JSON → POST.
* Escape prompt+context into s_work_small halves, then build JSON
* in s_work_large (no overlap). */
/* 1) Build context string in s_work_large */
size_t ctx_len = claw_session_build_context(s_work_large, sizeof(s_work_large) - 256);
/* 2) Escape prompt into s_work_small first half */
char *esc_p = s_work_small;
size_t ep_max = sizeof(s_work_small) / 2;
size_t ep_len = json_escape(esc_p, ep_max, prompt,
prompt_len > 200 ? 200 : prompt_len);
/* 3) Escape context into s_work_small second half */
char *esc_c = s_work_small + ep_max;
size_t ec_max = sizeof(s_work_small) / 2;
size_t ec_len = json_escape(esc_c, ec_max,
s_work_large, ctx_len > 200 ? 200 : ctx_len);
/* ...context: inside bentoclaw_ask() ... */
/* 4) Build JSON in s_work_large (no overlap with s_work_small) */
int json_len = snprintf(s_work_large, sizeof(s_work_large),
"{\"prompt\":\"%.*s\",\"context\":\"%.*s\"}",
(int)ep_len, esc_p, (int)ec_len, esc_c);

ข้อกำหนดการเรียกใช้
เขียน ring ที่ยังค้างอยู่ลงไฟล์ /.bentoclaw_session บน QSPI ต้องเรียกจาก MicroPython task context (เป็นข้อกำหนดใน header — การเขียนทำโดยรัน open() ของ Python) คืนค่า true เมื่อสำเร็จ; เมื่อล้มเหลว เซสชันจะยังคงค้างอยู่เพื่อให้การเรียกครั้งถัดไปลองใหม่ — ห้ามล้างสถานะรอบ ๆ การเขียนแฟลชที่ล้มเหลว การเขียนหน่วยความจำแฟลชกินเวลาระดับมิลลิวินาที: ให้เรียกในจังหวะที่ว่าง ไม่ใช่เรียกทุกข้อความ ไม่มีผู้เรียกที่มีอยู่จริงในของที่ส่งมอบ ที่ใดเลย

static void bento_ex_claw_session_flush(void)
{
if (!claw_session_dirty()) {
return; /* nothing pending */
}
if (!claw_session_flush()) {
/* Failed write: the session stays dirty — the next idle-point
* call retries. Do not clear anything here. */
return;
}
