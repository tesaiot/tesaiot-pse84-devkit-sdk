เป้าหมายของหัวข้อนี้

แยกให้ออกระหว่าง คำสั่ง กับ คำขอ stop() เป็นคำสั่ง unload() เป็นคำขอที่เครื่องยนต์ปฏิเสธได้ ส่วน stage_load() ไม่มีการตอบรับใด ๆ เลย ต้องยืนยันด้วยการเฝ้าดูจำนวนใน registry ว่าเพิ่มขึ้น ทั้ง 3 อย่างมีเส้นทางการตรวจสอบต่างกัน และบทนี้เดินครบทั้ง 3 เส้นทาง

ลำดับการทำงานของเฟิร์มแวร์

Stop edge_ai.stop() ส่งคำสั่ง STOP ผ่านการเชื่อมโมเดล แล้วจึงยืนยันสถานะว่าง (idle) แบบพยายามให้ดีที่สุด ฝั่ง CM55 เรียก ai_engine_stop() — ซึ่งเป็นคำขอ และ ai_engine_active() == -1 คือการยืนยัน — แล้วจึง ปรับอัตราเซนเซอร์ลงกลับสู่จังหวะของแดชบอร์ด:

ai_engine_stop(); /* a request, not yet a fact */
ai_engine_set_sensor_rate(100u); /* back to the 10 Hz dashboard cadence */
/* ... later: ai_engine_active() == -1 confirms the stop */
ai_engine_stop

หยุดโมเดลที่ active อยู่ (idempotent); ต้องจับคู่กับการลดอัตราเซนเซอร์ลง
ai_engine_set_sensor_rate

ตั้งช่วงเวลาการป้อนข้อมูลของเซนเซอร์วัดความเร่ง; ให้เรียกเดี่ยว ๆ และมีเจ้าของรายเดียว

การเรียกปรับอัตราเป็นข้อบังคับ: หากไม่เรียก accelerometer จะค้างอยู่ที่ 50 Hz และแย่งทรัพยากรจนเซนเซอร์ของแดชบอร์ดขาดแคลน

Unload edge_ai.unload(i) เป็น "คำขอ ไม่ใช่คำสั่ง" ให้ส่ง stop() ก่อน แล้วอ่าน edge_ai.diag() เพื่อดูว่าคำขอสำเร็จหรือถูกปฏิเสธ ฝั่ง C ใช้ตัวนับคู่เดียวกัน โดยเก็บค่าตั้งต้นไว้ ก่อนส่งคำขอ:

const uint32_t done0 = ai_engine_unload_done();
const uint32_t refused0 = ai_engine_unload_refused();
ai_engine_unload(idx); /* asynchronous */
/* ... later: done - done0 == 1 -> released; refused - refused0 == 1 -> refused */
ai_engine_unload
void ai_engine_unload(uint32_t idx)
คำขอ unload แบบอะซิงโครนัส — จะถูกปฏิเสธ เว้นแต่ engine ว่างอยู่

คำขอจะสำเร็จก็ต่อเมื่อเครื่องยนต์อยู่ในสถานะว่าง — การหยุดต้องถูก ยืนยันแล้ว (ai_engine_active() == -1) ไม่ใช่เพียงถูกร้องขอ — และโมเดลนั้นถูกโหลดอยู่และมี finalize() มิฉะนั้นคำขอจะถูกปฏิเสธและถูกนับไว้ หลังจาก unload แล้ว การ select โมเดลนั้นครั้งถัดไปจะโหลดมันใหม่ตั้งแต่ต้น

โมเดล staged edge_ai.stage_load(...) ส่ง MODEL_LINK_CMD_LOAD_STAGED และฝั่ง CM55 จัดการมันใน task context ไม่ใช่ใน callback ของ pipe เด็ดขาด เพราะ ai_model_staged_load() จองหน่วยความจำจาก heap ของ CM55 ซึ่งจอแสดงผลใช้ร่วมอยู่:

int idx = ai_model_staged_load(header_addr); /* task context only */
if (idx < 0) {
/* -1..-7, listed in ai_model_staged.h; ai_model_staged_last_rc() keeps it */
}

ไม่มีเหตุการณ์ตอบรับใด ๆ ฝั่ง MicroPython จึงยืนยันด้วยการเฝ้าดูจำนวนใน registry ว่าเพิ่มขึ้น

ทีละขั้น

ขั้นที่ 1 — หยุดโมเดลที่กำลังทำงาน

>>> edge_ai.select(0); edge_ai.stop()
>>> edge_ai.active()
-1

สิ่งที่ควรสังเกต active() คืนค่า −1 เมื่อการหยุดได้รับการยืนยันแล้ว บนหน้าจอ badge เปลี่ยนเป็น Stopped (ไม่ใช่ Failed เพราะ stack_words() ไม่เป็นศูนย์) ส่วนแถวเซนเซอร์ของแดชบอร์ดกลับสู่จังหวะปกติภายในราว 2 s

ขั้นที่ 2 — สั่ง unload ขณะที่ยังทำงานอยู่ แล้วดูการปฏิเสธ

>>> edge_ai.select(0)
True
>>> before = edge_ai.diag()['unload_refused']
>>> edge_ai.unload(0)
>>> edge_ai.diag()['unload_refused'] - before
1

สิ่งที่ควรสังเกต unload_refused เพิ่มขึ้นหนึ่ง และ unload_done ไม่ขยับ เพราะเครื่องยนต์ไม่ได้อยู่ในสถานะว่าง คำขอจึงถูกปฏิเสธและถูกนับไว้ ตัวนับนั้น คือคำตอบ

ขั้นที่ 3 — สั่ง unload หลังหยุดแล้ว แล้วดูการทำงานสำเร็จ

>>> edge_ai.stop()
>>> before = edge_ai.diag()['unload_done']
>>> edge_ai.unload(0)
>>> edge_ai.diag()['unload_done'] - before
1

สิ่งที่ควรสังเกต unload_done เพิ่มขึ้น การ select(0) ครั้งถัดไปจะเดินเส้นทางโหลดแบบเย็นเต็มรูปอีกครั้ง (นานถึง 15 s) ซึ่งเป็นผลที่ตั้งใจไว้ ให้วนอ่านคู่ unload_done/unload_refused ไปด้วยกัน การอ่านเพียงตัวเดียวไม่บอกอะไรเลย

ขั้นที่ 4 — stage โมเดลหนึ่งตัว แล้วยืนยันด้วยการสังเกต

>>> n0 = len(edge_ai.models())
>>> edge_ai.stage_load(...) # per the edge_ai module reference
>>> len(edge_ai.models()) - n0
1

สิ่งที่ควรสังเกต จำนวนใน registry เพิ่มขึ้นหนึ่ง นั่นคือการยืนยันเพียงอย่างเดียวที่มี หากไม่เพิ่ม ให้อ่าน edge_ai.diag() เพื่อดูตัวนับฝั่ง staged — staged_rejects และ staged_last_rc มาเป็นคู่ และ cm55_heap_free บอกพื้นที่ว่างของ heap บน CM55

ขั้นที่ 5 — ทำให้โมเดล staged เข้าร่วม set

>>> edge_ai.set_define(253, [0, n0]) # n0 = the new row's index

สิ่งที่ควรสังเกต การอ่านกลับด้วย Q_SET แสดงสมาชิกครบทั้ง 2 ตัว การเรียก set_define() ด้วยดัชนีใน registry ของแถวใหม่คือวิธีที่รองรับในการนำโมเดลที่อัปโหลดเข้ามาไปไว้ใน set อย่าพึ่งการตั้งชื่อให้ซ้ำกับสมาชิกที่คอมไพล์มา

Warning
ห้ามเรียก ai_engine_resume_sensor() ทันทีหลัง ai_engine_set_sensor_rate() คำขอทั้งสองทำงานแบบไม่ประสานเวลา และ resume ที่ออกไปขณะที่คำขอปรับอัตรายังค้างอยู่อาจยกเลิกคำขอนั้น CM33 จะหยุดการส่งข้อมูลเซนเซอร์แทนการปรับอัตรา ทั้งนี้ set_sensor_rate() กลับมาส่งข้อมูลเซนเซอร์ต่อให้เองอยู่แล้ว resume_sensor ไม่มีผู้เรียกที่ใดเลย ในเฟิร์มแวร์ที่ส่งมอบ ตัวอย่างที่เขียนขึ้นเองจึงเป็นการส่งแบบยืนเดี่ยว โดยมี anti-pattern กำกับไว้ในกรอบ:
__attribute__((used))
static void bento_ex_ai_engine_resume_sensor(void)
{
/* WRONG — the two requests can cancel each other (see the box):
*
* ai_engine_set_sensor_rate(100u); // rate request, in flight...
* ai_engine_resume_sensor(); // ...can cancel it -> PAUSE
*
* RIGHT — set_sensor_rate alone already resumes the sensor push:
*
* ai_engine_set_sensor_rate(100u); // walk down to dashboard rate
*
* RIGHT — resume on its own, with no rate change in flight
* (page-leave path, no rate change wanted): */
ai_engine_resume_sensor();
}
Warning
ให้การปรับอัตราเซนเซอร์มีเจ้าของรายเดียว หาก task สองตัวต่างเรียก set_sensor_rate()/resume_sensor() โดยไม่รู้กัน คำขอก็ยกเลิกกันเองได้แบบเดียวกัน เฟิร์มแวร์ที่ส่งมอบตัดสินอัตราจากจุดเดียว โค้ดของผู้อ่านก็ควรทำเช่นนั้น
การหยุดต้องถูกยืนยัน ไม่ใช่เพียงถูกร้องขอ unload() ที่ออกตามหลัง stop() ในจังหวะเดียวกันมักถูกปฏิเสธ เพราะเครื่องยนต์ยังไม่ได้ยืนยันการหยุด ให้ตรวจ active() == -1 ก่อน
