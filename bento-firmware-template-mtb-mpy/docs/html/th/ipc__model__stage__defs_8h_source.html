4/* Staging a downloaded model for CM55 — the contract both cores compile.
5 *
6 * ── Why a shared buffer and not an IPC stream ─────────────────────────────
7 * A model arrives on CM33 over TACP and lands on LittleFS. CM55 cannot read
8 * it there: the filesystem volume is 0x60C00000-0x63FFFFFF and CM55's only
9 * external-flash grant is 0x60580000-0x60BBFFFF (cycfg_system.c, M55_mpc_regions),
10 * so not one byte overlaps — and CM55 has no filesystem driver at all
11 * (LV_USE_FS_* are 0). The widest reliable CM33->CM55 hop in this tree is 240
12 * bytes, and bento_link's data plane is `return false;`.
13 *
14 * So the bytes travel through memory both cores already map. m33_m55_shared
15 * (0x262FC000, 256 KB) is granted RW to PC5 (CM33_NS) and PC6 (CM55_NS) on the
16 * same SOCMEM MPC block that already serves .ml_weights, and CM55's MPU region
17 * covering it is Normal NON-CACHEABLE — ARM_MPU_ATTR_NON_CACHEABLE is the value
18 * 4 that cycfg_system.c stores in a field named `cacheable`, which reads as the
19 * opposite of what it means. CM33 has no data cache at all (__DCACHE_PRESENT 0).
20 * Both facts together are why this needs no cache maintenance, and they are the
21 * same reason the existing IPC windows need none.
22 *
23 * ── Why the parameters ride in the buffer, not in the message ─────────────
24 * MODEL_LINK_OP_CTRL carries one command byte and one uint32. A model needs a
25 * dozen numbers plus a name and its class labels. Rather than invent a
26 * multi-message parameter protocol, the manifest sits at the head of the shared
27 * buffer and the message carries only its address. One command, one pointer,
28 * and CM55 reads the rest from memory it can already see.
29 *
30 * ── Alignment ─────────────────────────────────────────────────────────────
31 * The Ethos-U driver rejects a command stream or base address that is not
32 * 16-byte aligned (ethosu_driver.c, MASK_16_BYTE_ALIGN) and Vela pads the
33 * command stream relative to the FLATBUFFER BASE, so the flatbuffer itself must
34 * be aligned. We require 32 rather than 16: the CM55 cache-maintenance granule
35 * is 32 bytes (__SCB_DCACHE_LINE_SIZE), and while this region is non-cacheable
36 * today, a buffer that is exclusive on both granules cannot be broken by a
37 * future MPU change. 32 satisfies both and costs at most 31 bytes.
38 *
39 * Alignment is NOT inherited from the linker. .cy_shared_socmem is declared
40 * ALIGN(4) and its base in the current link is 0x26301A3C, which is 12 mod 16 —
41 * the section's alignment is a function of HSM symbol sizes in an unrelated
42 * translation unit. The staging code aligns its own pointer at run time and
43 * reports it, rather than trusting a section. */

101/* The width of ONE frame the IMU path actually hands a model: feed_imu() builds
102 * a `float in[6]` -- accel xyz then gyro xyz -- on its stack and passes that.
103 *
104 * A staged model declares its own frame width in the manifest, and until this
105 * constant existed nothing compared the two. The check that did exist,
106 * window_depth * frame_bytes == in_elements * 4, is an identity between three
107 * declared numbers: it is satisfied by a consistent LIE. A genuine 9-channel
108 * export -- accel + gyro + magnetometer, and this board carries a BMM350 --
109 * declares 9 floats per frame, passes that identity, passes the input-tensor
110 * check because the tensor really does have that many elements, and then the
111 * copy reads 36 bytes out of a 24-byte stack array, fifty times a second.
112 *
113 * No adversary is needed for that: bento_deepcraft_import.py reads
114 * frame_floats straight out of the DEEPCRAFT export, so the supported
115 * toolchain produces it from an honest model. Nothing crashes -- the read is
116 * in bounds of the stack frame, just past the array -- so the failure is
117 * silently wrong answers, which is the worst shape this system has. */

138/* Manifest, at offset 0 of the shared staging buffer. The flatbuffer follows at
139 * model_offset, which must itself be a multiple of AI_STAGE_ALIGN.
140 *
141 * Every field here is something a downloaded .tflite cannot tell us:
142 *
143 * - arena_bytes. NOT derivable from the file. Measured on this hardware
144 * 2026-08-10: the Motion model's own OfflineMemoryAllocation extent is
145 * 18,192 bytes and initialising with it FAILS (rc -2); the true floor is
146 * between 18,192 and 19,336. A loader that reads the plan and trusts it is
147 * wrong every time. Declare it -- or leave it 0 and CM55 will walk a ladder
148 * of sizes until one takes. Probing is safe because an undersized arena was
149 * measured to fail cleanly: error returned, engine idle, REPL alive, no
150 * fault. This is the one required field a human can safely forget.
151 * - the window geometry. A .tflite describes a tensor, not how a sensor
152 * stream becomes one. Motion's is frame 24 B, depth 100, stride 10.
153 * - class labels, which live in the generated C, not the model.
154 *
155 * This is the honest boundary of what this path can load: models whose
156 * preprocessing is "slide a window, then quantize". Motion qualifies. Audio and
157 * radar do NOT — their DSP graphs (Hann window, RFFT, mel filterbank, bin
158 * selection) and kilobytes of coefficient tables are C code that no .tflite
159 * carries. Do not extend this struct to pretend otherwise; a real answer there
160 * is a described pipeline, not more fields. */

208/* ── Signature ─────────────────────────────────────────────────────────────
209 * A TRAILER, after the flatbuffer, not a header field.
210 *
211 * The header is 504 bytes and the flatbuffer starts at 512 -- numbers that are
212 * now proven on hardware, including the 32-byte alignment the Ethos-U requires
213 * of a Vela command stream. Adding a field would move every one of them and
214 * re-open a question that has been answered. A trailer changes nothing that
215 * already works: an unsigned blob is byte-identical to what shipped before,
216 * and the board tells them apart by looking past the model for the magic.
217 *
218 * CRC32 answers "did this arrive intact". It answers nothing about who sent
219 * it -- anyone can recompute a CRC over anything. That distinction is the
220 * whole reason this exists: mtb_ml_model_init() validates NOTHING (the shipped
221 * libtensorflow-microlite.a exports zero Verify* symbols), so a malformed blob
222 * becomes wild pointer arithmetic inside AllocateTensors on the core that owns
223 * the display, and CM55's fault handler disables interrupts and blinks until
224 * someone pulls power. A signature is what makes "the store sent this" a
225 * checkable claim rather than an assumption.
226 *
227 * ECDSA P-256 over SHA-256: the curve OPTIGA Trust M does in hardware, and the
228 * one this project already signs and verifies with elsewhere.
229 *
230 * The signature covers every byte from offset 0 to the start of this trailer
231 * -- manifest and model together. Signing only the model would leave the
232 * window geometry, the training rate and the axis convention unauthenticated,
233 * and those are precisely the fields whose wrongness cannot be detected at
234 * run time. */
235#define AI_STAGE_SIG_MAGIC (0x47495342u) /* 'BSIG' */
236#define AI_STAGE_SIG_ALG_ECDSA_P256 (1u)
237
238typedef struct {
239 uint32_t magic; /* AI_STAGE_SIG_MAGIC */

241 uint8_t sig[64]; /* raw r||s, 32 bytes each */
242 uint8_t key_id[8]; /* which signing key, for rotation */
243 uint32_t reserved;
244 uint32_t footer_crc32; /* over every byte above this field */
245} ai_stage_sig_t;

249/* Where the trailer sits, given a manifest. Word-aligned so the CRC over it

251static inline uint32_t ai_stage_sig_offset(const ai_stage_header_t *h)

256/* CRC-32/ISO-HDLC (reflected 0xEDB88320), the same polynomial the IDE already
257 * uses. Table-free: this runs once per upload on CM33 and once per load on
258 * CM55, so a 1 KB table would cost more than the loop saves.
259 *
260 * static inline in the shared header so both cores compute it from ONE source.
261 * Two implementations of a checksum is how a checksum stops meaning anything. */
262static inline uint32_t ai_stage_crc32(uint32_t crc, const uint8_t *p, uint32_t n)
263{
264 crc = ~crc;
265 for (uint32_t i = 0u; i < n; i++) {
266 crc ^= p[i];
267 for (uint32_t b = 0u; b < 8u; b++) {
268 crc = (crc >> 1) ^ (0xEDB88320u & (uint32_t)(-(int32_t)(crc & 1u)));
269 }
270 }
271 return ~crc;
272}
