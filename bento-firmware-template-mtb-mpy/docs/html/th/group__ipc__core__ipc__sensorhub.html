ต่อหนึ่ง tick มีผู้อ่านได้รายเดียว — การอ่านจะเคลียร์แฟล็ก changed

Take a snapshot of the latest sensor data. Clears the "changed" flags after reading. Safe to call from any task context.

Parameters
snapOutput snapshot struct.
ข้อกำหนดการเรียกใช้
ผู้เรียกเป็นเจ้าของ sensorhub_snapshot_t บน stack ของตนเอง และส่งที่อยู่ของมันเข้ามา header ระบุว่า "Safe to call from any task context" แต่ระบุด้วยว่า "Clears the 'changed' flags after reading" — ดังนั้นต่อหนึ่ง tick จึงมีผู้อ่านเรียกได้ เพียงรายเดียว พอดี นั่นคือเหตุผลที่ UI อ่าน snapshot ครั้งเดียวที่ 33 ms แล้วกระจายต่อผ่าน pm_render() แทนที่จะให้แต่ละหน้า (page) อ่าน snapshot ของตัวเองแยกกัน ผู้เรียกที่อยู่ใน archive อีก 2 รายซึ่งอยู่ในบริบทอื่นยืนยันข้ออ้างเรื่อง any-task นี้: AI inference task (ai_engine.c:795) และ IPC service task (ipc_service.c:729) คืนค่า void
variant ที่ใช้ได้
mtb-mpy และ mtb-only
static void sensorhub_timer_cb(lv_timer_t *timer)
{
(void)timer;
/* Take snapshot of all sensor data from IPC */
sensorhub_snapshot_t snap;
ipc_sensorhub_snapshot(&snap);
/* Dispatch render to current page (skips if animating) */
pm_render(&s_pm, &snap);
/* Update global topbar (time + WiFi) on current page */
if (!s_pm.animating) {

}
}

bool ipc_sensorhub_wifi_connected ( void )

สถานะ WiFi ที่ได้จากการวนถาม — ต้อง edge-detect ก่อนแตะ LVGL

Returns true if CM33_NS reported WiFi connected via IPC_CMD_WIFI_STATE_PUSH

ข้อกำหนดการเรียกใช้
ได้มาจากการวนถาม (polling) — ไม่มี API แบบ callback อยู่เลย รูปแบบ edge-detect (ตรวจเฉพาะขอบที่สถานะเปลี่ยน) เป็นข้อบังคับ: ให้เทียบสถานะที่วนถามได้กับแฟล็ก LVGL ปัจจุบันของ widget แล้วแตะ LVGL เฉพาะเมื่อมีการเปลี่ยนสถานะจริงเท่านั้น การเรียก add_flag / clear_flag ทุก tick โดยไม่มีเงื่อนไขทำให้แถวการ์ดกะพริบจนมองเห็นได้ มีตัวกัน (guard) กรณี widget เป็น null เพราะหน้า (page) อาจถูกทำลายระหว่าง tick ทำงานที่การจำกัดอัตรา (throttle) ของ topbar ประมาณ 6 fps และถูกข้ามระหว่างที่มีแอนิเมชัน
variant ที่ใช้ได้
mtb-mpy และ mtb-only
/* ...context: inside pm_update_topbar() - GFX task context ... */
/* WiFi icon — only toggle visibility when state actually changes
* to prevent unnecessary LVGL invalidation (card row flicker). */

bool connected = ipc_sensorhub_wifi_connected();
bool hidden = lv_obj_has_flag(pm->wifi_lbl, LV_OBJ_FLAG_HIDDEN);

lv_obj_clear_flag(pm->wifi_lbl, LV_OBJ_FLAG_HIDDEN);
} else if (!connected && !hidden) {
lv_obj_add_flag(pm->wifi_lbl, LV_OBJ_FLAG_HIDDEN);
}
}

ตัวอย่าง (เขียนขึ้นเอง — ไม่มี call site ในของที่ส่งมอบจริง)
static lv_obj_t *s_ble_lbl; /* created by the page's create_cb,
* NULLed by its destroy_cb */
__attribute__((used))
static void bento_ex_ipc_sensorhub_ble_connected(void)
{
if (s_ble_lbl == NULL) {
return; /* page gone — widget freed */
}
bool connected = ipc_sensorhub_ble_connected();
bool hidden = lv_obj_has_flag(s_ble_lbl, LV_OBJ_FLAG_HIDDEN);
/* Edge detect: touch LVGL only on a state CHANGE. */
if (connected && hidden) {
lv_obj_clear_flag(s_ble_lbl, LV_OBJ_FLAG_HIDDEN);
} else if (!connected && !hidden) {
lv_obj_add_flag(s_ble_lbl, LV_OBJ_FLAG_HIDDEN);
}
}

เป็น gate ของการอ่าน RTC ทั้งหมด

Returns true if CM33_NS has sent at least one IPC_CMD_TIME_PUSH (NTP synced)

Returns true after CM33_NS sends IPC_CMD_NTP_SYNCED (RTC is valid).

ข้อกำหนดการเรียกใช้

bool ipc_sensorhub_get_time_str ( char * buf,
size_t buf_size )

สตริงเวลาล่าสุดที่ CM33 ส่งมา; คืน false ก่อน NTP sync

Copy the latest time string from CM33_NS (e.g. "Mon 10 Mar 14:35"). Returns false if NTP not yet synced. buf must be >= 32 bytes.

ข้อกำหนดการเรียกใช้
คัดลอกสตริงเวลาล่าสุดที่ CM33_NS ส่งมา และคืนค่า false เมื่อ NTP ยังไม่ sync buf ต้องมีขนาดอย่างน้อย 32 ไบต์ ไม่มีผู้เรียกที่มีอยู่จริงในของที่ส่งมอบ — topbar อ่าน RTC โดยตรงหลัง gate ntp_synced แทน ตัวอย่างที่เขียนขึ้นเองจับคู่ฟังก์ชันนี้เข้ากับ gate ดังกล่าว
variant ที่ใช้ได้
mtb-mpy และ mtb-only
ตัวอย่าง (เขียนขึ้นเอง — ไม่มี call site ในของที่ส่งมอบจริง)
__attribute__((used))
static void bento_ex_ipc_sensorhub_get_time_str(void)
{
/* Gate first: false before the first NTP push — necessary, and this
* call re-checks it anyway (returns false while unsynced). */
if (!ipc_sensorhub_ntp_synced()) {
return; /* show nothing rather than garbage */
}
char now[32]; /* header contract: >= 32 bytes */
if (!ipc_sensorhub_get_time_str(now, sizeof(now))) {
return;
}
/* Drive a label only on change (flicker discipline, GFX task):
* if (strcmp(lv_label_get_text(lbl), now) != 0)
* lv_label_set_text(lbl, now);
*/
(void)now;
}
